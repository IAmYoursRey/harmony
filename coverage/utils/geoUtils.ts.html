export const getDistanceFromLatLonInKm = (lat1: number, lon1: number, lat2: number, lon2: number) => {
  const R = 6371; // Radius of the earth in km
  const dLat = deg2rad(lat2 - lat1);  
  const dLon = deg2rad(lon2 - lon1); 
  const a = 
    Math.sin(dLat/2) * Math.sin(dLat/2) +
    Math.cos(deg2rad(lat1)) * Math.cos(deg2rad(lat2)) * 
    Math.sin(dLon/2) * Math.sin(dLon/2)
    ; 
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)); 
  const d = R * c; // Distance in km
  return d;
}

export const distancePointToLineSegmentKm = (
  pLat: number, pLng: number,
  A_lat: number, A_lng: number,
  B_lat: number, B_lng: number
): number => {
  const x0 = pLng, y0 = pLat;
  const x1 = A_lng, y1 = A_lat;
  const x2 = B_lng, y2 = B_lat;
 
  // Convert to kilometers (approximate for short distances)
  const dx = x2 - x1;
  const dy = y2 - y1;
 
  if (dx === 0 && dy === 0) {
    return getDistanceFromLatLonInKm(pLat, pLng, A_lat, A_lng);
  }
 
  // Calculate t that minimizes the distance
  let t = ((x0 - x1) * dx + (y0 - y1) * dy) / (dx * dx + dy * dy);
  t = Math.max(0, Math.min(1, t));
 
  const closestX = x1 + t * dx;
  const closestY = y1 + t * dy;
 
  return getDistanceFromLatLonInKm(pLat, pLng, closestY, closestX);
};
